# 帧缓冲与后处理管线

[[toc]]

---

## 一、为什么需要后处理

传统的前向渲染（Forward Rendering）把光照计算直接做在物体绘制上，有两大局限：
1. **光源数量受限** — 每多一个光源，着色器复杂度线性增加
2. **全局效果难实现** — Bloom、SSAO、景深、运动模糊需要整帧信息

后处理（Post-Processing）先把场景渲染到纹理（FBO），再对这个纹理做各种图像处理。

---

## 二、帧缓冲对象（FBO）

### 2.1 概念

```
默认帧缓冲（屏幕）
    ↓
自定义帧缓冲（纹理）
    - 颜色附件（Color Attachment）→ 纹理或 renderbuffer
    - 深度附件（Depth Attachment）
    - 模板附件（Stencil Attachment）
```

### 2.2 Three.js 中创建 FBO

```js
import * as THREE from 'three';

// 创建渲染目标（Render Target）
const renderTarget = new THREE.WebGLRenderTarget(
    window.innerWidth,
    window.innerHeight,
    {
        minFilter: THREE.LinearFilter,
        magFilter: THREE.LinearFilter,
        format: THREE.RGBAFormat,
        type: THREE.UnsignedByteType,  // 或 HalfFloatType 用于 HDR
        depthBuffer: true,
        stencilBuffer: false
    }
);

// 渲染场景到 FBO
renderer.setRenderTarget(renderTarget);
renderer.render(scene, camera);
renderer.setRenderTarget(null);  // 切回屏幕

// renderTarget.texture 即为场景帧
```

### 2.3 Ping-Pong 缓冲

很多效果需要多次迭代，用两个 FBO 交替读写：

```js
const fboA = new THREE.WebGLRenderTarget(w, h);
const fboB = new THREE.WebGLRenderTarget(w, h);

let readFBO = fboA, writeFBO = fboB;

for (let i = 0; i < passes; i++) {
    renderer.setRenderTarget(writeFBO);
    postProcessMaterial.uniforms.uInputTexture.value = readFBO.texture;
    renderer.render(postProcessScene, orthoCamera);

    // 交换
    [readFBO, writeFBO] = [writeFBO, readFBO];
}
```

## 三、常用后处理效果

### 3.1 高斯模糊（Gaussian Blur）

用于 Bloom、软阴影、DOF 等。分为水平和垂直两次一维卷积，复杂度从 O(n²) 降到 O(2n)。

```glsl
// 水平高斯模糊
uniform sampler2D uTexture;
uniform vec2 uResolution;
uniform float uBlurRadius;

const float weights[5] = float[](0.227027, 0.1945946, 0.1216216, 0.054054, 0.016216);

vec4 gaussianBlur(sampler2D image, vec2 uv, vec2 direction) {
    vec4 color = texture2D(image, uv) * weights[0];
    vec2 off = direction / uResolution * uBlurRadius;

    for (int i = 1; i < 5; i++) {
        color += texture2D(image, uv + off * float(i)) * weights[i];
        color += texture2D(image, uv - off * float(i)) * weights[i];
    }
    return color;
}

void main() {
    gl_FragColor = gaussianBlur(uTexture, vUv, vec2(1.0, 0.0));  // 水平
    // 第二次 pass 用 vec2(0.0, 1.0) 做垂直
}
```

### 3.2 Bloom（辉光）

步骤：
1. 渲染场景到 HDR 缓冲
2. 提取亮部分（设置阈值）
3. 对亮部分做高斯模糊（通常多层，从半分辨率开始）
4. 将模糊后的亮部分叠加回原图

```glsl
// 亮部提取
vec4 extractBright(vec3 color, float threshold) {
    float brightness = dot(color, vec3(0.2126, 0.7152, 0.0722));
    return brightness > threshold ? vec4(color, 1.0) : vec4(0.0);
}

// 最终合成
vec3 bloomComposite(vec3 original, vec3 bloom, float intensity) {
    return original + bloom * intensity;
}
```

### 3.3 色调映射（Tone Mapping）

HDR 值超出 [0,1] 范围，需要映射到 LDR 显示范围。

```glsl
// Reinhard 映射（简单）
vec3 reinhard(vec3 hdr) {
    return hdr / (1.0 + hdr);
}

// 更保留高光细节的 Reinhard 变体
vec3 reinhardExtended(vec3 hdr, float maxWhite) {
    return hdr * (1.0 + hdr / (maxWhite * maxWhite)) / (1.0 + hdr);
}

// ACES 映射（影视标准）
vec3 acesFilmic(vec3 x) {
    float a = 2.51;
    float b = 0.03;
    float c = 2.43;
    float d = 0.59;
    float e = 0.14;
    return clamp((x * (a * x + b)) / (x * (c * x + d) + e), 0.0, 1.0);
}

// Unreal 的 ACES approximation
vec3 unrealACES(vec3 x) {
    return clamp((x * (0.6 * x + 0.04)) / (x * (0.6 * x + 0.44) + 0.05), 0.0, 1.0);
}
```

### 3.4 Gamma 校正

```glsl
// sRGB 空间转换
vec3 linearToSRGB(vec3 linear) {
    return pow(linear, vec3(1.0 / 2.2));
}

vec3 sRGBToLinear(vec3 srgb) {
    return pow(srgb, vec3(2.2));
}
```

### 3.5 SSAO（屏幕空间环境光遮蔽）

基于深度缓冲计算 AO，不需要第二次场景渲染。

```glsl
uniform sampler2D uDepthTexture;
uniform sampler2D uNormalTexture;
uniform mat4 uProjectionMatrix;
uniform vec2 uResolution;

// 将深度转换为视图空间坐标
vec3 reconstructViewPos(vec2 uv, float depth) {
    vec4 clip = vec4(uv * 2.0 - 1.0, depth * 2.0 - 1.0, 1.0);
    vec4 view = inverse(uProjectionMatrix) * clip;
    return view.xyz / view.w;
}

float ssao(vec2 uv, vec3 viewPos, vec3 normal) {
    float occlusion = 0.0;
    float radius = 0.5;
    int samples = 16;

    for (int i = 0; i < 16; i++) {
        // 在法线半球上采样随机点
        vec3 sampleDir = normalize(hash33(vec3(uv, float(i))) * 2.0 - 1.0);
        sampleDir = faceNormal(sampleDir, normal);

        vec3 samplePos = viewPos + sampleDir * radius;
        vec4 offset = uProjectionMatrix * vec4(samplePos, 1.0);
        offset.xy = offset.xy / offset.w * 0.5 + 0.5;

        float sampleDepth = texture2D(uDepthTexture, offset.xy).r;
        vec3 sampleViewPos = reconstructViewPos(offset.xy, sampleDepth);

        float rangeCheck = smoothstep(0.0, 1.0, radius / abs(viewPos.z - sampleViewPos.z));
        occlusion += (sampleViewPos.z < samplePos.z ? 1.0 : 0.0) * rangeCheck;
    }
    return 1.0 - (occlusion / float(samples));
}
```

### 3.6 景深（Depth of Field）

```glsl
// 简单景深：根据深度调整模糊强度
float getCoC(float depth, float focusDistance, float focusRange) {
    return clamp((depth - focusDistance) / focusRange, -1.0, 1.0);
}

vec3 depthOfField(sampler2D colorTex, sampler2D depthTex, vec2 uv,
                  float focusDistance, float focusRange) {
    float depth = texture2D(depthTex, uv).r;
    float coc = abs(getCoC(depth, focusDistance, focusRange));

    vec3 color = texture2D(colorTex, uv).rgb;
    vec3 blurred = gaussianBlur(colorTex, uv, vec2(1.0, 0.0), coc * 10.0);
    blurred = gaussianBlur(colorTex, uv, vec2(0.0, 1.0), coc * 10.0);

    return mix(color, blurred, coc);
}
```

## 四、MRT（Multiple Render Targets）

一次绘制输出多个缓冲，用于延迟渲染（Deferred Rendering）。

```js
// WebGL2 / Three.js 中
const gBuffer = new THREE.WebGLRenderTarget(width, height, {
    count: 3  // 3 个颜色附件
});

// 片段着色器
const gBufferMaterial = new THREE.ShaderMaterial({
    glslVersion: THREE.GLSL3,
    fragmentShader: `
        layout(location = 0) out vec4 gAlbedo;
        layout(location = 1) out vec4 gNormal;
        layout(location = 2) out vec4 gPosition;

        void main() {
            gAlbedo = vec4(albedo, 1.0);
            gNormal = vec4(normalize(vNormal) * 0.5 + 0.5, 1.0);
            gPosition = vec4(vWorldPos, 1.0);
        }
    `
});
```

## 五、完整示例：Three.js 后处理管线

```js
import * as THREE from 'three';
import { EffectComposer } from 'three/examples/jsm/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/examples/jsm/postprocessing/RenderPass.js';
import { ShaderPass } from 'three/examples/jsm/postprocessing/ShaderPass.js';

// 自定义 Bloom + Tone Mapping Shader
const CustomPostProcess = {
    uniforms: {
        tDiffuse: { value: null },
        uBloomIntensity: { value: 0.5 },
        uExposure: { value: 1.0 }
    },
    vertexShader: `
        varying vec2 vUv;
        void main() { vUv = uv; gl_Position = vec4(position, 1.0); }
    `,
    fragmentShader: `
        uniform sampler2D tDiffuse;
        uniform float uBloomIntensity;
        uniform float uExposure;
        varying vec2 vUv;

        vec3 acesFilmic(vec3 x) {
            return clamp((x * (2.51 * x + 0.03)) / (x * (2.43 * x + 0.59) + 0.14), 0.0, 1.0);
        }

        void main() {
            vec3 color = texture2D(tDiffuse, vUv).rgb;

            // 提取亮部（简化 bloom）
            float luma = dot(color, vec3(0.2126, 0.7152, 0.0722));
            vec3 bloom = color * smoothstep(0.8, 1.5, luma) * uBloomIntensity;

            // 曝光 + 色调映射
            vec3 hdr = (color + bloom) * uExposure;
            vec3 ldr = acesFilmic(hdr);

            // Gamma
            gl_FragColor = vec4(pow(ldr, vec3(1.0 / 2.2)), 1.0);
        }
    `
};

const composer = new EffectComposer(renderer);
composer.addPass(new RenderPass(scene, camera));
composer.addPass(new ShaderPass(CustomPostProcess));

// 渲染循环
function animate() {
    requestAnimationFrame(animate);
    composer.render();
}
```

## 六、奇淫技巧

### 6.1 半浮点 HDR 缓冲

```glsl
// 用 RGBE 格式压缩 HDR 数据（不需要 WebGL2 HalfFloat）
vec4 encodeRGBE(vec3 rgb) {
    float maxComponent = max(max(rgb.r, rgb.g), rgb.b);
    float exponent = ceil(log2(maxComponent));
    return vec4(rgb / exp2(exponent), (exponent + 128.0) / 255.0);
}

vec3 decodeRGBE(vec4 rgbe) {
    float exponent = rgbe.a * 255.0 - 128.0;
    return rgbe.rgb * exp2(exponent);
}
```

### 6.2 双线性过滤优化

```glsl

vec4 cubic(float v) {

    float z = s.z - 4.0 * s.y + 6.0 * s.x;
    float w = 6.0 - x - y - z;
```

### 6.3 缩小彩色渲染

```glsl
// 展示 3 个 mip level 的内容
vec3 showMipmap(sampler2D tex, vec2 uv) {
    vec3 col0 = texture2D(tex, uv).rgb;
```
