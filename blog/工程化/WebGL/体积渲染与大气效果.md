## 一、为什么体积渲染很难

传统渲染只计算**表面**。但云、雾、烟、大气是**参与介质**（Participating Media），光线在穿过它们时会被吸收、散射、发射。

```
入射光 I(0)  →  [体积介质]  →  出射光 I(d)
                 ↓ 散射
              各个方向的光
```

---

## 二、Beer-Lambert 定律

光线穿过吸收介质时的衰减：

```
I(d) = I(0) * exp(-∫₀ᵈ σₜ(p) ds)
```

- `σₜ` = 消光系数（吸收 + 散射）
- `d` = 穿过介质的距离

```glsl
// 简化的光学厚度计算
float opticalDepth = density * extinction * distance;
float transmittance = exp(-opticalDepth);
```

## 三、Ray Marching 体积云

### 3.1 基础框架

```glsl
#define MAX_STEPS 64
#define ABSORPTION 0.5
#define SCATTERING 0.3

float cloudDensity(vec3 p) {
    // 基础形状：高度衰减 + FBM 噪声
    float heightFactor = smoothstep(0.0, 1.0, (p.y - cloudBottom) / (cloudTop - cloudBottom));
    heightFactor *= 1.0 - smoothstep(0.7, 1.0, heightFactor);  // 顶部消减

    float noise = fbm(p * 0.5 + vec3(uTime * 0.1, 0.0, uTime * 0.05), 5);
    float density = max(0.0, noise - 0.4) * heightFactor * 2.0;
    return density;
}

vec4 renderCloud(vec3 ro, vec3 rd) {
    float t = intersectBox(ro, rd, cloudBoundsMin, cloudBoundsMax);
    if (t < 0.0) return vec4(0.0);

    vec3 p = ro + t * rd;
    float stepSize = length(cloudBoundsMax - cloudBoundsMin) / float(MAX_STEPS);

    vec3 color = vec3(0.0);
    float transmittance = 1.0;

    for (int i = 0; i < MAX_STEPS; i++) {
        float density = cloudDensity(p);

        if (density > 0.001) {
            // 朝光源采样（单次散射近似）
            float lightTransmittance = exp(-cloudDensity(p + sunDir * 0.5) * 10.0);
            vec3 scattering = sunColor * lightTransmittance * SCATTERING;

            // 内散射 + 外散射
            float dT = exp(-density * ABSORPTION * stepSize);
            color += scattering * transmittance * (1.0 - dT);
            transmittance *= dT;

            if (transmittance < 0.01) break;
        }

        p += rd * stepSize;
    }

    return vec4(color, 1.0 - transmittance);
}
```

### 3.2 廉价的体积光（God Rays）

```glsl
// 从屏幕空间深度缓冲计算体积光
float godRays(vec2 uv, vec2 lightPosUV) {
    vec2 delta = (lightPosUV - uv) / float(SAMPLES);
    vec2 pos = uv;
    float illumination = 0.0;

    for (int i = 0; i < SAMPLES; i++) {
        pos += delta;
        float depth = texture2D(uDepth, pos).r;
        float occlusion = step(depth, 0.999);  // 粗略判断是否在阴影中
        illumination += occlusion;
    }

    illumination /= float(SAMPLES);
    return illumination * exp(-length(lightPosUV - uv) * 3.0);
}
```

## 四、大气散射

### 4.1 Rayleigh 散射

空气分子对短波长（蓝紫光）散射更强，这就是为什么天空是蓝色的。

```glsl
// Rayleigh 散射系数 ~ λ⁻⁴
vec3 rayleighCoefficient = vec3(5.8e-6, 13.5e-6, 33.1e-6);

float rayleighPhase(float cosTheta) {
    return (3.0 / (16.0 * PI)) * (1.0 + cosTheta * cosTheta);
}
```

### 4.2 Mie 散射

气溶胶粒子对所有波长散射相近，产生白色的雾和光晕。

```glsl
// Mie 散射具有强前向性
float miePhase(float cosTheta, float g) {
    float g2 = g * g;
    return (1.0 / (4.0 * PI)) *
           ((1.0 - g2) / pow(1.0 + g2 - 2.0 * g * cosTheta, 1.5));
}
```

### 4.3 简化大气

```glsl
vec3 atmosphere(vec3 r, vec3 ro, vec3 sunDir) {
    // r: 视线方向, ro: 观察点位置, sunDir: 太阳光方向
    float sunDot = max(dot(r, sunDir), 0.0);

    // Rayleigh
    vec3 rayleigh = vec3(0.3, 0.5, 1.0) * (1.0 + sunDot * sunDot);

    // Mie
    float mie = pow(sunDot, 32.0) * 2.0;

    // 地平线衰减
    float horizon = 1.0 - abs(r.y);
    horizon = exp(-horizon * 3.0);

    return rayleigh * horizon + vec3(mie) * horizon;
}
```

## 五、完整示例：程序化天空盒

```glsl
precision highp float;
uniform vec3 uSunDir;
uniform float uTime;
varying vec3 vDirection;

#define PI 3.14159265359

float hash(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453);
}

float valueNoise(vec2 p) {
    vec2 i = floor(p);
    vec2 f = fract(p);
    f = f * f * (3.0 - 2.0 * f);
    return mix(mix(hash(i), hash(i + vec2(1,0)), f.x),
               mix(hash(i + vec2(0,1)), hash(i + vec2(1,1)), f.x), f.y);
}

float fbm(vec2 p, int octaves) {
    float v = 0.0, a = 0.5;
    for (int i = 0; i < 5; i++) {
        if (i >= octaves) break;
        v += a * valueNoise(p); p *= 2.0; a *= 0.5;
    }
    return v;
}

// 简化云密度
float cloudDensity(vec3 p) {
    if (p.y < 0.2 || p.y > 0.8) return 0.0;
    float h = smoothstep(0.2, 0.5, p.y) * smoothstep(0.8, 0.5, p.y);
    vec2 cloudUV = p.xz / (p.y + 0.1);
    float n = fbm(cloudUV * 2.0 + uTime * 0.02, 4);
    return max(0.0, n - 0.5) * h * 3.0;
}

void main() {
    vec3 rd = normalize(vDirection);
    vec3 sunDir = normalize(uSunDir);

    // 大气
    float sunDot = max(dot(rd, sunDir), 0.0);
    float horizon = 1.0 - abs(rd.y);
    horizon = exp(-horizon * 3.0);

    vec3 sky = mix(vec3(0.0, 0.05, 0.15), vec3(0.3, 0.6, 1.0), rd.y * 0.5 + 0.5);
    sky += vec3(1.0, 0.9, 0.7) * pow(sunDot, 256.0) * horizon;  // 太阳盘
    sky += vec3(0.5, 0.6, 0.8) * pow(sunDot, 8.0) * horizon;    // Mie 光晕

    // 云
    float t = 0.0;
    vec3 cloudColor = vec3(0.0);
    float transmittance = 1.0;

    for (int i = 0; i < 32; i++) {
        vec3 p = rd * (1.0 + float(i) * 0.05);
        float d = cloudDensity(p);
        if (d > 0.001) {
            float light = exp(-cloudDensity(p + sunDir * 0.3) * 5.0);
            vec3 scatter = vec3(1.0, 0.98, 0.95) * light + vec3(0.1);
            float dT = exp(-d * 0.5);
            cloudColor += scatter * transmittance * (1.0 - dT);
            transmittance *= dT;
            if (transmittance < 0.01) break;
        }
    }

    vec3 finalColor = mix(sky, cloudColor, 1.0 - transmittance);
    gl_FragColor = vec4(pow(finalColor, vec3(1.0 / 2.2)), 1.0);
}
```

## 六、奇淫技巧

### 6.1 蓝噪声采样减少步进条纹

```glsl
// 用蓝噪声偏移步进起点，打破带状伪影
float blueNoise = texture2D(uBlueNoise, gl_FragCoord.xy / 64.0).r;
vec3 p = ro + rd * (t + blueNoise * stepSize);
```

### 6.2 自适应步进

```glsl
// 密度高的地方步进更密
float adaptiveStep = baseStep / max(density * 2.0, 0.5);
```

### 6.3 低分辨率体素缓冲

```glsl
// 把体积密度预计算到 3D 纹理或低分辨率缓冲
// 然后采样时插值
float density = texture3D(uVoxelTex, worldPos / volumeSize).r;
```

### 6.4 假体积雾

```glsl
// 不需要 ray marching 的廉价雾
float fog = exp(-max(length(viewPos) - fogStart, 0.0) * fogDensity);
// 或者基于高度的雾
```
